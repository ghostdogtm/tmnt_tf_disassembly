enum ; fighter phase (objPhase of slots 0/1), handled by FighterPhase
PHASE_SPAWN = 0 ; place at the start position, full health
PHASE_READY = 1 ; one frame: sets the intro timer
PHASE_INTRO = 2 @objPhase ; waiting for the round to start
PHASE_FIGHT = 3 @objPhase ; objState runs
PHASE_KO = 4 @objPhase ; knocked out / round lost (objState = step)
endenum

enum ; fighter action state (objState while PHASE_FIGHT), handled by FighterState
FST_STAND = 0 ; on the floor: walk, turn, read buttons
FST_JUMP = 1
FST_SHOT = 2 @objState ; crouch pose, then throws the ball (object slot 6: ballOwner, fighterHasBall, SND_SHOT); Down held: FST_CROUCH, else stand
FST_HURT = 3 @objState ; hit reaction, kind in objHitType
FST_KNOCKDOWN = 4 @objState ; thrown up, falls, lies, gets up (buttons speed it up)
FST_SPECIAL = 5 @objState ; animated special move (SpecialScriptStart / SpecialFxFrames); its effect object is slot $0C + fighter
FST_TURN = 6 @objState ; turns round to face the opponent
FST_CROUCH = 7 @objState
FST_CHAR_MOVE = 8 @objState ; per-character move (jump table on objSpriteSet)
FST_HOLD = 9 @objState ; holds the opponent (throw / repeated hits)
FST_HELD = $0A @objState ; held by the opponent, loses health on each hit
FST_0B = $0B @objState ; caught in the opponent's character move (State0B: blinking frame, SND_STATE_0B, not pushed apart by SeparateFighters; ShredderBoltsWait waits for it); back to air hurt when the opponent's FST_CHAR_MOVE step ends; not seen in the Mesen runs
FST_WIN = $0C @objState ; victory pose and voice
endenum

SC_LEN = $D0 ; + base length; then channel setup: square 3 bytes (envelope, duty/volume, decay), triangle 1 (decay), drums 0
SC_PACKED = $DE ; toggle packed notes: two pitches per byte (high, low nibble), each one base length

SC_ENV = $E6 ; + env byte: bit 7 set = fade mode (low nibble = sndFadeSteps), else an EnvelopePtrs id (stored *2)
SC_RELEASE = $E7 ; + release: the note is cut after n/16 of its length (0 = never)
SC_LENGTH = $E8 ; + base length
SC_DUTY = $E9 ; + n: low nibble 0 = duty bits (sndDuty, ORed into the volume register); else flags (bit 1: the high nibble of n replaces the duty/volume bits); 0 = clear the flags
SC_FADE = $EA ; + n: fade-in at each note start: high nibble = starting volume, low nibble = frames per volume step (0 = no fade-in)
SC_VIBRATO = $EB ; 0 = off, else a, b: sndVibId = (a-$50)*2 (pitch envelope), b = sndVibDepth: low nibble = depth (offset step * depth), high nibble = delay before the vibrato starts (x base length)
SC_FLAGS = $EC ; + n: bits 7-6 = sndVolMin (lowest volume), bits 5-4 -> sndEchoFlags bits 3-2, bits 3-0 = echo mode (0 off; 1-3: n | $A0; 4-F: (n-3) | $B0); the echo replays the period of an earlier note from the channel's 4-note ring (EchoStore / EchoPeriodLoad)
SC_DETUNE = $ED ; + pitch offset
SC_TRANSPOSE = $EE ; + note offset
SC_DIVIDER = $EF ; + n: every n-th frame the music channels skip their update (slower tempo), 0 = off (musicSkipPeriod)
SC_VOLUME = $F0 ; + volume offset 0-8
SC_SLIDE = $FA ; open a repeat block: passes, step[, volume ramp]; each pass plays the block again with the period lowered by another `step & $7F` (16-bit, accumulating); square / noise: every (b >> 4) passes the volume index grows by (b & 7), max $F (CmdSlide; read from code, not heard)
SC_SLIDE_END = $FA ; the next $FA: jump back to the block start while passes are left, then close the block
SC_LOOP = $FB ; loop start
SC_LOOP2 = $FC ; loop start, second level
SC_CALL = $FD ; + word: call a sub-stream (one level)
SC_REPEAT = $FE ; + count: back to the loop start count times
SC_GOTO = $FF ; SC_REPEAT, SC_GOTO + word: jump
SC_END = $FF ; end of the stream / return from SC_CALL
